Option Strict On Option Explicit On Imports System.Collections.Generic Imports System.Web.UI.WebControls Public NotInheritable Class GridHelper Private Sub New() End Sub Public Shared Function ColumnIndex(grid As GridView, field As String) As Integer For i As Integer = 0 To grid.Columns.Count - 1 Dim col As DataControlField = grid.Columns(i) Dim bound As BoundField = TryCast(col, BoundField) If bound IsNot Nothing AndAlso String.Equals(bound.DataField, field, StringComparison.OrdinalIgnoreCase) Then Return i Dim link As HyperLinkField = TryCast(col, HyperLinkField) If link IsNot Nothing AndAlso String.Equals(link.DataTextField, field, StringComparison.OrdinalIgnoreCase) Then Return i If String.Equals(col.SortExpression, field, StringComparison.OrdinalIgnoreCase) Then Return i Next Return -1 End Function Public Shared Sub AddCellClass(grid As GridView, row As GridViewRow, field As String, cssClass As String) Dim i As Integer = ColumnIndex(grid, field) If i < 0 OrElse i >= row.Cells.Count Then Return Dim cell As TableCell = row.Cells(i) If String.IsNullOrEmpty(cell.CssClass) Then cell.CssClass = grid.Columns(i).ItemStyle.CssClass AddClass(cell, cssClass) End Sub Public Shared Sub AddClass(control As WebControl, cssClass As String) Dim current As String = If(control.CssClass, "").Trim() If (" " & current & " ").Contains(" " & cssClass & " ") Then Return control.CssClass = (current & " " & cssClass).Trim() End Sub Public Shared Sub PrepareHeader(grid As GridView, sortKey As String, sortDir As String) If grid.HeaderRow Is Nothing Then Return grid.HeaderRow.TableSection = TableRowSection.TableHeader If String.IsNullOrEmpty(sortKey) Then Return For i As Integer = 0 To grid.Columns.Count - 1 If String.Equals(grid.Columns(i).SortExpression, sortKey, StringComparison.OrdinalIgnoreCase) Then Dim cell As TableCell = grid.HeaderRow.Cells(i) If String.IsNullOrEmpty(cell.CssClass) Then cell.CssClass = grid.Columns(i).HeaderStyle.CssClass AddClass(cell, If(sortDir = "DESC", "sort-desc", "sort-asc")) End If Next End Sub Public Shared Function SortText(fields As String, direction As String) As String Dim parts As New List(Of String) For Each f As String In fields.Split(","c) If f.Trim() <> "" Then parts.Add(f.Trim() & " " & If(direction = "DESC", "DESC", "ASC")) Next Return String.Join(", ", parts) End Function Public Shared Function IsBlank(value As Object) As Boolean Return value Is Nothing OrElse Convert.IsDBNull(value) OrElse Convert.ToString(value).Trim() = "" End Function Public Shared Function PagerItems(pageIndex As Integer, pageCount As Integer) As List(Of PagerItem) Dim items As New List(Of PagerItem) If pageCount <= 1 Then Return items items.Add(New PagerItem(ChrW(171), If(pageIndex > 0, pageIndex - 1, -1), "Previous page")) Dim pages As New SortedSet(Of Integer) From {0, pageCount - 1} For p As Integer = pageIndex - 2 To pageIndex + 2 If p >= 0 AndAlso p < pageCount Then pages.Add(p) Next Dim previous As Integer = -1 For Each p As Integer In pages If previous >= 0 AndAlso p - previous > 1 Then items.Add(New PagerItem(ChrW(8230), -1, "")) Dim item As New PagerItem((p + 1).ToString(), p, "Page " & (p + 1).ToString()) item.IsActive = (p = pageIndex) items.Add(item) previous = p Next items.Add(New PagerItem(ChrW(187), If(pageIndex < pageCount - 1, pageIndex + 1, -1), "Next page")) Return items End Function End Class Public Class PagerItem Public Sub New(text As String, index As Integer, title As String) Me.Text = text Me.Index = index Me.Title = title End Sub Public Property Text As String Public Property Index As Integer Public Property Title As String Public Property IsActive As Boolean Public ReadOnly Property IsLink As Boolean Get Return Index >= 0 AndAlso Not IsActive End Get End Property Public ReadOnly Property CssClass As String Get If IsActive Then Return "page-item active" If Index < 0 Then Return "page-item disabled" Return "page-item" End Get End Property End Class
